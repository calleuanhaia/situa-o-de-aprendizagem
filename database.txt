CREATE DATABASE MEJDATABASE2;
USE MEJDATABASE2;

CREATE TABLE Cliente (
	id_cliente INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    nome_cliente VARCHAR(100),
    cidade VARCHAR(50),
    uf VARCHAR(2),
    nome_contato VARCHAR(100),
    email VARCHAR(100),
    segmento VARCHAR(50),
    status VARCHAR(50)
);

CREATE TABLE Funcionario (
	id_funcionario INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    nome_funcionario VARCHAR(100),
    cargo VARCHAR(50),
    departamento VARCHAR(50),
    data_admissao DATE,
    salario DECIMAL(10,2),
    gestor VARCHAR(50)
);

CREATE TABLE Projeto (
	id_projeto INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    nome_projeto VARCHAR(50),
    id_cliente INT NOT NULL,
    id_funcionario INT NOT NULL,
    data_inicio DATE,
    data_entrega DATE,
    orcamento DECIMAL(10,2),
    status VARCHAR(50),
    modulo VARCHAR(50),
    
    CONSTRAINT fk_id_cliente_cliente FOREIGN KEY(id_cliente) REFERENCES Cliente (id_cliente),
    CONSTRAINT fk_id_funcionario_funcionario FOREIGN KEY(id_funcionario) REFERENCES Funcionario (id_funcionario)
);

CREATE TABLE Chamado (
	id_chamado INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    id_cliente INT NOT NULL,
    id_funcionario INT NOT NULL,
    data_abertura DATE,
    categoria VARCHAR(50),
    prioridade VARCHAR(50),
    descricao VARCHAR(255),
    status VARCHAR(50),
    horas_gastas DOUBLE,
    
    CONSTRAINT fk_id_cliente_cliente2 FOREIGN KEY(id_cliente) REFERENCES Cliente (id_cliente),
    CONSTRAINT fk_id_funcionario_funcionario2 FOREIGN KEY(id_funcionario) REFERENCES Funcionario (id_funcionario)
);

CREATE TABLE OP (
	id_op INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    id_cliente INT NOT NULL,
    produto VARCHAR(100),
    quantidade INT,
    unidade VARCHAR(2),
    id_maquina INT,
    data_inicio DATE,
    data_fim DATE,
    prioridade VARCHAR(50),
    status VARCHAR(50),
    
    CONSTRAINT fk_id_cliente_cliente3 FOREIGN KEY(id_cliente) REFERENCES cliente (id_cliente)
);

CREATE TABLE apontamento (
	id_apontamento INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    id_op INT NOT NULL,
    id_maquina INT,
    data DATE,
    turno VARCHAR(10),
    quantidade_produzida INT,
    unidade VARCHAR(2),
    horas_paradas DOUBLE,
    motivo_parada VARCHAR(100),
    operador VARCHAR(50),
    
    CONSTRAINT fk_id_op_op FOREIGN KEY(id_op) REFERENCES op (id_op)
);
